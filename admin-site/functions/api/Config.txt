// functions/api/config.js
// ══════════════════════════════════════════════════════════════════
// Cloudflare Pages Function — Integration credentials API
// Reads & writes all credentials to KV (UP_STORE_DATA namespace).
//
// SETUP (one-time in Cloudflare dashboard):
//   Workers & Pages → your project → Settings → Bindings
//   → Add → KV namespace
//   Variable name:  UP_STORE_DATA
//   KV namespace:   UP_STORE_DATA  (the existing one)
//   → Save and redeploy
//
// GET  /api/config  — returns saved credentials as JSON
// POST /api/config  — saves credentials from JSON body
// ══════════════════════════════════════════════════════════════════

const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
  'Content-Type': 'application/json',
};

const CONFIG_KEY = 'up:integrations';

const ALLOWED_KEYS = [
  'paypalClientId',
  'emailjsPublicKey',
  'emailjsServiceId',
  'emailjsTemplateId',
  'emailjsDesignTemplateId',
  'emailjsQuoteTemplateId',
  'emailjsQuoteErinTemplateId',
  'emailjsOrderConfirmTemplateId',
  'emailjsNewOrderTemplateId',
  'emailjsOrderReadyTemplateId',
  'cfAccountId',
  'cfImagesToken',
];

export async function onRequestGet(context) {
  try {
    const raw = await context.env.UP_STORE_DATA.get(CONFIG_KEY);
    const config = raw ? JSON.parse(raw) : {};
    return new Response(JSON.stringify({ ok: true, config }), { headers: CORS });
  } catch (e) {
    return new Response(JSON.stringify({ ok: false, error: e.message }), {
      status: 500, headers: CORS,
    });
  }
}

export async function onRequestPost(context) {
  try {
    const body = await context.request.json();
    const raw = await context.env.UP_STORE_DATA.get(CONFIG_KEY);
    const existing = raw ? JSON.parse(raw) : {};
    const updated = { ...existing };

    for (const key of ALLOWED_KEYS) {
      if (body[key] !== undefined && body[key] !== null) {
        // Empty string clears the value
        if (body[key] === '') delete updated[key];
        else updated[key] = String(body[key]).trim();
      }
    }

    await context.env.UP_STORE_DATA.put(CONFIG_KEY, JSON.stringify(updated));
    return new Response(JSON.stringify({ ok: true, config: updated }), { headers: CORS });
  } catch (e) {
    return new Response(JSON.stringify({ ok: false, error: e.message }), {
      status: 500, headers: CORS,
    });
  }
}

export async function onRequestOptions() {
  return new Response(null, { status: 204, headers: CORS });
}
